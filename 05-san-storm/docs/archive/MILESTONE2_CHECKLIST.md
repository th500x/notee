```javascript
// 使用localStorage存储收藏数据
const STORAGE_KEY = 'san_storm_favorites';

// 收藏武将
function addFavorite(characterId, tags = []) {
  const favorites = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
  favorites[characterId] = {
    addedAt: Date.now(),
    tags: tags,
  };
  localStorage.setItem(STORAGE_KEY, JSON.stringify(favorites));
}

// 获取收藏列表
function getFavorites() {
  return JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
}

// 添加标签
function addTag(characterId, tag) {
  const favorites = getFavorites();
  if (favorites[characterId]) {
    if (!favorites[characterId].tags.includes(tag)) {
      favorites[characterId].tags.push(tag);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(favorites));
    }
  }
}
```
